112 friend std::ostream&
113 operator<<(std::ostream& os, const ElementaryFile& ef) {
114 /*
115 * The SFI is formatted apart: streamed as a uint8_t it would come out
116 * as a character, and most SFI values are not printable.
117 */
118 std::ostringstream sfi;
119 sfi << std::uppercase << std::hex << std::setfill('0') << std::setw(2)
120 << static_cast<int>(ef.getSfi());
121
122 os << "ELEMENTARY_FILE: {"
123 << "SFI: " << sfi.str() << "h";
124
125 /*
126 * A header is reported as present only. FileHeader includes this header
127 * for the Type enumeration, so it is an incomplete type here and cannot
128 * be rendered; print the header itself to see its fields.
129 */
130 if (ef.getHeader() != nullptr) {
131 os << ", HEADER: set";
132 }
133
134 os << ", " << ef.getData() << "}";
135
136 return os;
137 }
